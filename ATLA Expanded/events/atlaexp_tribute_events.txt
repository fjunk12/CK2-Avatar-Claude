# ATLA Expanded - Ba Sing Se tribute system
# atlaexp.1  yearly collection, runs once a year on the ruler of Ba Sing Se (on_yearly_pulse, hidden)
# atlaexp.2  the Earth King calls an obedient realm to defend against a foreign attacker
# atlaexp.3  "Ba Sing Se and us": shows a ruler their current obligation (from a decision)
# atlaexp.4  a foreign or Way of Chin holder of Ba Sing Se loots the treasury (on_yearly_pulse, hidden)
# atlaexp.5  the Royal Treasury: balance and projects (from a decision)
# atlaexp.6  a great work is finished: every tribute-owing ruler reacts (AI and players)
# atlaexp.7  resentment: a player ruler is invited to stop paying
# atlaexp.8  the Royal Messenger Network collapses (upkeep not paid)
# atlaexp.9  the Royal Roads fall into disrepair (upkeep not paid)
namespace = atlaexp

character_event = {
	id = atlaexp.1
	hide_window = yes
	is_triggered_only = yes # on_yearly_pulse

	only_independent = yes

	trigger = {
		OR = {
			has_landed_title = e_earth_kingdom
			AND = {
				has_landed_title = k_earth_kingdom
				NOT = { e_earth_kingdom = { has_holder = yes } }
			}
		}
		atx_valid_earth_king_trigger = yes
	}

	immediate = {
		save_event_target_as = atx_earth_king
		109 = { save_event_target_as = atx_bss } # Royal Earth Palace
		# Graft: an unreformed bureaucracy loses 10% of the treasury a year, a reformed one 3%
		if = {
			limit = { has_character_modifier = atx_crown_bureaucracy }
			multiply_variable = { which = global_atx_treasury value = 0.97 }
		}
		else = {
			multiply_variable = { which = global_atx_treasury value = 0.9 }
		}
		atx_refresh_projects_effect = yes
		atx_great_works_upkeep_effect = yes
		# The states slowly forget their grievances
		change_variable = { which = global_atx_resentment value = -5 }
		atx_clamp_resentment_effect = yes
		clr_character_flag = atx_under_foreign_attack
		if = {
			limit = {
				any_war = {
					defender = { character = ROOT }
					attacker = { NOT = { religion_group = earth_religions } }
				}
			}
			set_character_flag = atx_under_foreign_attack
		}
		any_independent_ruler = {
			limit = {
				atx_tribute_payer_base_trigger = yes
				NOT = { war_with = ROOT }
				atx_tier_recognised_trigger = yes
			}
			atx_find_regional_lord_effect = yes
			if = {
				limit = { has_character_flag = atx_withholding }
				# Defiance hurts the crown only where it actually expects tribute
				if = {
					limit = {
						atx_tier_tribute_trigger = yes
						higher_tier_than = COUNT
					}
					change_variable = { which = global_royalAuthority value = -1 }
					prestige = 5
					ROOT = { opinion = { who = PREV modifier = opinion_atx_withholds_tribute months = 13 } }
				}
			}
			else_if = {
				limit = { atx_tier_tribute_trigger = yes }
				atx_pay_tribute_effect = yes
				if = {
					limit = { has_global_flag = atx_work_law }
					opinion = { who = ROOT modifier = opinion_atx_crown_laws months = 13 }
				}
				# Resentment: some states stop paying
				if = {
					limit = { check_variable = { which = global_atx_resentment value = 25 } }
					random = {
						chance = 4
						modifier = { factor = 2 check_variable = { which = global_atx_resentment value = 50 } }
						modifier = { factor = 2 check_variable = { which = global_atx_resentment value = 75 } }
						modifier = { factor = 2 trait = proud }
						modifier = { factor = 2 trait = ambitious }
						modifier = { factor = 0.25 opinion = { who = ROOT value = 25 } }
						atx_start_grumbling_effect = yes
					}
				}
				if = {
					limit = {
						ROOT = { has_character_flag = atx_under_foreign_attack }
						atx_tier_obedience_trigger = yes
						NOT = { war_with = ROOT }
						OR = {
							NOT = { has_character_flag = atx_called }
							had_character_flag = { flag = atx_called days = 700 }
						}
					}
					clr_character_flag = atx_called
					set_character_flag = atx_called
					letter_event = { id = atlaexp.2 days = 3 }
				}
			}
			else = {
				atx_pay_recognised_effect = yes
			}
			clr_character_flag = atx_has_regional
		}
	}
}

letter_event = {
	id = atlaexp.2
	desc = atlaexp.2.d
	border = GFX_event_letter_frame_war
	is_triggered_only = yes

	trigger = {
		FROM = { is_alive = yes }
		NOT = { war_with = FROM }
	}

	option = { # Answer the call
		name = atlaexp.2.a
		join_defender_wars = FROM
		FROM = { opinion = { who = ROOT modifier = opinion_atx_answered_call years = 10 } }
		ai_chance = {
			factor = 70
			modifier = { factor = 2 opinion = { who = FROM value = 25 } }
			modifier = { factor = 0.3 NOT = { opinion = { who = FROM value = 0 } } }
			modifier = { factor = 0.5 trait = craven }
			modifier = { factor = 1.5 trait = brave }
		}
	}
	option = { # Refuse
		name = atlaexp.2.b
		prestige = -50
		change_variable = { which = global_royalAuthority value = -3 }
		FROM = { opinion = { who = ROOT modifier = opinion_atx_refused_call years = 10 } }
		ai_chance = { factor = 30 }
	}
}

# A great work is finished. FROM = the ruler of Ba Sing Se. Each tribute-owing ruler chooses to accept or resist.
character_event = {
	id = atlaexp.6
	desc = {
		trigger = { has_character_flag = atx_news_messengers }
		text = atlaexp.6.d_messengers
	}
	desc = {
		trigger = { has_character_flag = atx_news_roads }
		text = atlaexp.6.d_roads
	}
	desc = {
		trigger = { has_character_flag = atx_news_law }
		text = atlaexp.6.d_law
	}
	desc = {
		trigger = { has_character_flag = atx_news_wall }
		text = atlaexp.6.d_wall
	}
	picture = GFX_evt_ba_sing_se
	is_triggered_only = yes

	option = { # Accept
		name = atlaexp.6.a
		clr_character_flag = atx_news_messengers
		clr_character_flag = atx_news_roads
		clr_character_flag = atx_news_law
		clr_character_flag = atx_news_wall
		ai_chance = {
			factor = 70
			modifier = { factor = 2 opinion = { who = FROM value = 25 } }
			modifier = { factor = 1.5 trait = content }
		}
	}
	option = { # Resist
		name = atlaexp.6.b
		trigger = { NOT = { has_character_flag = atx_withholding } }
		clr_character_flag = atx_news_messengers
		clr_character_flag = atx_news_roads
		clr_character_flag = atx_news_law
		clr_character_flag = atx_news_wall
		prestige = 10
		set_character_flag = atx_withholding
		add_character_modifier = { name = atx_defying_ba_sing_se duration = -1 }
		ai_chance = {
			factor = 30
			modifier = { factor = 2 trait = proud }
			modifier = { factor = 2 trait = ambitious }
			modifier = { factor = 0.3 opinion = { who = FROM value = 25 } }
		}
	}
}
