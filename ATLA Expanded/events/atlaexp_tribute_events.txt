# ATLA Expanded - Ba Sing Se: tribute, treasury, works, help requests, sabotage, Dai Li
# atlaexp.1   yearly run on the ruler of Ba Sing Se (on_yearly_pulse, hidden)
# atlaexp.2   the Earth King calls an obedient realm to defend against a foreign attacker
# atlaexp.3   "Ba Sing Se and Us": a ruler's current obligation (decision)
# atlaexp.4   a foreign or Way of Chin holder of Ba Sing Se loots the treasury (on_yearly_pulse, hidden)
# atlaexp.5   the Royal Treasury overview (decision)
# atlaexp.6   a great work is finished: every tribute-owing ruler accepts or resists
# atlaexp.7   resentment: a player ruler is invited to stop paying
# atlaexp.8   the Royal Messenger Network collapses (upkeep unpaid)
# atlaexp.9   the Royal Roads fall into disrepair (upkeep unpaid)
# atlaexp.10  a work completes a stage
# atlaexp.11  a work stalls or is abandoned
# atlaexp.12  a far realm asks Ba Sing Se for help (Earth King decides)
# atlaexp.13  hidden hop on the requesting realm (sets the request type, forwards it)
# atlaexp.14  the requesting realm hears the answer
# atlaexp.15  a work is completed
# atlaexp.16  a budget programme lapses (unpaid)
# atlaexp.17  the Dai Li overreach
# atlaexp.18  sabotage at the works (culprit unknown)
# atlaexp.19  sabotage foiled (FROM = the saboteur)
namespace = atlaexp

character_event = {
	id = atlaexp.1
	hide_window = yes
	is_triggered_only = yes # on_yearly_pulse

	only_independent = yes

	trigger = {
		atx_ruler_of_ba_sing_se_trigger = yes
	}

	immediate = {
		log = "ATLA Expanded: yearly Ba Sing Se run for [Root.GetFullName]"
		save_event_target_as = atx_earth_king
		109 = { save_event_target_as = atx_bss } # Royal Earth Palace
		atx_compute_effective_authority_effect = yes
		set_variable = { which = global_atx_last_income value = 0 }
		set_variable = { which = global_atx_last_payers value = 0 }
		set_variable = { which = global_atx_last_owing value = 0 }
		atx_crown_revenue_effect = yes

		# Graft: 10% a year, 3% with the bureaucracy programme, 5% more when the Dai Li skim the treasury
		if = {
			limit = { has_global_flag = atx_prog_bureaucracy }
			multiply_variable = { which = global_atx_treasury value = 0.97 }
		}
		else = {
			multiply_variable = { which = global_atx_treasury value = 0.9 }
		}
		if = {
			limit = { check_variable = { which = global_atx_daili value = 50 } }
			multiply_variable = { which = global_atx_treasury value = 0.95 }
		}

		atx_programmes_effect = yes
		atx_great_works_upkeep_effect = yes
		atx_progress_all_works_effect = yes
		atx_daili_influence_effect = yes

		# Resentment fades, except under Monarchism ("worshipped above any Spirit")
		if = {
			limit = { has_global_flag = atx_law_monarchism }
			change_variable = { which = global_atx_resentment value = -3 }
		}
		else = {
			change_variable = { which = global_atx_resentment value = -5 }
		}
		atx_clamp_resentment_effect = yes

		clr_character_flag = atx_under_foreign_attack
		if = {
			limit = {
				any_war = {
					defender = { character = ROOT }
					attacker = { NOT = { religion_group = earth_religions } }
				}
			}
			set_character_flag = atx_under_foreign_attack
		}

		set_variable = { which = global_atx_requests value = 0 }

		any_independent_ruler = {
			limit = {
				atx_tribute_payer_base_trigger = yes
				NOT = { war_with = ROOT }
				atx_tier_recognised_trigger = yes
			}
			change_variable = { which = global_atx_last_owing value = 1 }
			atx_find_regional_lord_effect = yes
			if = {
				limit = { has_character_flag = atx_withholding }
				# Defiance hurts the crown only where it actually expects tribute
				if = {
					limit = {
						atx_tier_tribute_trigger = yes
						higher_tier_than = COUNT
					}
					change_variable = { which = global_royalAuthority value = -1 }
					prestige = 5
					ROOT = { opinion = { who = PREV modifier = opinion_atx_withholds_tribute months = 13 } }
				}
			}
			else_if = {
				limit = { atx_tier_tribute_trigger = yes }
				atx_pay_tribute_effect = yes
				if = {
					limit = { has_global_flag = atx_work_law }
					opinion = { who = ROOT modifier = opinion_atx_crown_laws months = 13 }
				}
				# Resentment: some states stop paying
				if = {
					limit = { check_variable = { which = global_atx_resentment value = 25 } }
					random = {
						chance = 4
						modifier = { factor = 2 check_variable = { which = global_atx_resentment value = 50 } }
						modifier = { factor = 2 check_variable = { which = global_atx_resentment value = 75 } }
						modifier = { factor = 2 trait = proud }
						modifier = { factor = 2 trait = ambitious }
						modifier = { factor = 0.25 opinion = { who = ROOT value = 25 } }
						atx_start_grumbling_effect = yes
					}
				}
				if = {
					limit = {
						ROOT = { has_character_flag = atx_under_foreign_attack }
						atx_tier_obedience_trigger = yes
						NOT = { war_with = ROOT }
						OR = {
							NOT = { has_character_flag = atx_called }
							had_character_flag = { flag = atx_called days = 700 }
						}
					}
					clr_character_flag = atx_called
					set_character_flag = atx_called
					letter_event = { id = atlaexp.2 days = 3 }
				}
			}
			else = {
				atx_pay_recognised_effect = yes
			}
			# Far realms that recognise the crown sometimes ask it for help (at most two requests a year)
			if = {
				limit = {
					atx_near_trigger = no
					NOT = { has_character_flag = atx_withholding }
					NOT = { check_variable = { which = global_atx_requests value = 2 } }
					OR = {
						NOT = { has_character_flag = atx_requested }
						had_character_flag = { flag = atx_requested days = 1800 }
					}
				}
				random = {
					chance = 3
					change_variable = { which = global_atx_requests value = 1 }
					clr_character_flag = atx_requested
					set_character_flag = atx_requested
					character_event = { id = atlaexp.13 }
				}
			}
			clr_character_flag = atx_has_regional
		}
		log = "ATLA Expanded: crown revenue [global_atx_last_crown.GetValue], tribute [global_atx_last_income.GetValue] gold from [global_atx_last_payers.GetValue] of [global_atx_last_owing.GetValue] owing realms. Treasury [global_atx_treasury.GetValue], Royal Authority [global_royalAuthority.GetValue], effective [global_atx_eff_ra.GetValue]"
	}
}
