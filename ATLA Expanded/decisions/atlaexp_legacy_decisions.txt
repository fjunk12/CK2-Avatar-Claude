	atx_proclaim_earth_empire = {
		is_high_prio = yes
		ai_check_interval = 12
		potential = {
			has_character_flag = atx_champion_earth
			independent = yes
			NOT = { is_title_active = e_earth_empire }
			NOR = {
				has_landed_title = e_earth_empire
				has_landed_title = e_earth_kingdom
			}
		}
		allow = {
			has_landed_title = c_royal_earth_palace
			OR = {
				completely_controls_region = world_north_east_earth_kingdom
				realm_size = 150
			}
			prestige = 500
		}
		effect = {
			prestige = -500
			primary_title = { save_event_target_as = atx_old_primary }
			e_earth_empire = {
				activate_title = { title = THIS status = yes } # e_earth_empire has allow = always no, so it is granted here
				grant_title = ROOT
				hidden_tooltip = {
					copy_title_laws = event_target:atx_old_primary
					make_primary_title = yes
				}
			}
			if = {
				limit = { has_landed_title = k_earth_kingdom }
				k_earth_kingdom = { de_jure_liege = e_earth_empire }
			}
			set_global_flag = atx_earth_empire_proclaimed
			any_player = {
				limit = { NOT = { character = ROOT } }
				character_event = { id = atlaexp.40 days = 3 }
			}
			# Ambition has had its way: the burden ends, the Empire remains
			clr_global_flag = atx_leg_ambition
			atx_end_champion_effect = yes
		}
