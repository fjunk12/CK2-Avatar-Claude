# ATLA Expanded - Ba Sing Se tribute system: effects
# Run in the payer's scope from atlaexp.1, with event targets atx_earth_king and atx_bss saved.
# If the payer's capital lies in a de jure kingdom held by another independent Earth ruler of king
# tier or higher, that regional ruler collects it for the crown and gains prestige.

atx_find_regional_lord_effect = {
	clr_character_flag = atx_has_regional
	save_event_target_as = atx_payer
	capital_scope = {
		kingdom = {
			if = {
				limit = {
					has_holder = yes
					holder_scope = {
						NOT = { character = event_target:atx_payer }
						NOT = { character = event_target:atx_earth_king }
						independent = yes
						higher_tier_than = DUKE
						atx_valid_earth_king_trigger = yes
					}
				}
				holder_scope = { save_event_target_as = atx_regional }
				event_target:atx_payer = { set_character_flag = atx_has_regional }
			}
		}
	}
}

# Pays N gold into the Ba Sing Se treasury (global_atx_treasury), never to the Earth King personally.
# A regional king collecting it on the crown's behalf earns prestige, not gold.
atx_pay_2_effect = {
	wealth = -2
	change_variable = { which = global_atx_treasury value = 2 }
	if = { # A Single Law for the Kingdom: half as much again
		limit = { has_global_flag = atx_work_law wealth = 1 }
		wealth = -1
		change_variable = { which = global_atx_treasury value = 1 }
	}
	if = {
		limit = { has_character_flag = atx_has_regional }
		event_target:atx_regional = { prestige = 2 }
	}
}

# Recognised: dukes 2, kings 4, emperors 8 (counts only acknowledge)
atx_pay_recognised_effect = {
	if = {
		limit = { tier = EMPEROR wealth = 8 }
		atx_pay_8_effect = yes
	}
	else_if = {
		limit = { tier = KING wealth = 4 }
		atx_pay_4_effect = yes
	}
	else_if = {
		limit = { tier = DUKE wealth = 2 }
		atx_pay_2_effect = yes
	}
}

# Tribute: dukes 4, kings 8, emperors 16. Counts pay 2 only under obedience.
atx_pay_tribute_effect = {
	if = {
		limit = { tier = EMPEROR wealth = 16 }
		atx_pay_16_effect = yes
	}
	else_if = {
		limit = { tier = KING wealth = 8 }
		atx_pay_8_effect = yes
	}
	else_if = {
		limit = { tier = DUKE wealth = 4 }
		atx_pay_4_effect = yes
	}
	else_if = {
		limit = { tier = COUNT wealth = 2 atx_tier_obedience_trigger = yes }
		atx_pay_2_effect = yes
	}
}

# Moves the whole treasury into the current scope's personal wealth (used when a foreign or Chin ruler
# takes Ba Sing Se and loots it). CK2 cannot add a variable to wealth, so it is paid out in chunks.
atx_loot_treasury_effect = {
	while = {
		limit = { check_variable = { which = global_atx_treasury value = 50 } }
		wealth = 50
		change_variable = { which = global_atx_treasury value = -50 }
	}
	while = {
		limit = { check_variable = { which = global_atx_treasury value = 10 } }
		wealth = 10
		change_variable = { which = global_atx_treasury value = -10 }
	}
	while = {
		limit = { check_variable = { which = global_atx_treasury value = 1 } }
		wealth = 1
		change_variable = { which = global_atx_treasury value = -1 }
	}
	set_variable = { which = global_atx_treasury value = 0 }
}

# Resentment of the states towards Ba Sing Se (global_atx_resentment, kept between 0 and 100)
atx_clamp_resentment_effect = {
	if = {
		limit = { NOT = { check_variable = { which = global_atx_resentment value = 0 } } }
		set_variable = { which = global_atx_resentment value = 0 }
	}
	if = {
		limit = { check_variable = { which = global_atx_resentment value = 100 } }
		set_variable = { which = global_atx_resentment value = 100 }
	}
}

# A payer joins the grumbling: the AI simply starts withholding, a player is asked
atx_start_grumbling_effect = {
	if = {
		limit = { ai = yes }
		set_character_flag = atx_withholding
		add_character_modifier = { name = atx_defying_ba_sing_se duration = -1 }
	}
	else = {
		letter_event = { id = atlaexp.7 }
	}
}

# Upkeep of finished great works, paid from the treasury or the work collapses (run on the ruler of Ba Sing Se)
atx_great_works_upkeep_effect = {
	if = {
		limit = { has_global_flag = atx_work_messengers }
		if = {
			limit = { check_variable = { which = global_atx_treasury value = 20 } }
			change_variable = { which = global_atx_treasury value = -20 }
		}
		else = {
			clr_global_flag = atx_work_messengers
			character_event = { id = atlaexp.8 }
		}
	}
	if = {
		limit = { has_global_flag = atx_work_roads }
		if = {
			limit = { check_variable = { which = global_atx_treasury value = 15 } }
			change_variable = { which = global_atx_treasury value = -15 }
		}
		else = {
			clr_global_flag = atx_work_roads
			any_province = {
				limit = { has_province_modifier = atx_royal_road }
				remove_province_modifier = atx_royal_road
			}
			character_event = { id = atlaexp.9 }
		}
	}
}

# Saves the ruler of Ba Sing Se as event_target:atx_earth_king (from any scope)
atx_save_earth_king_effect = {
	if = {
		limit = { e_earth_kingdom = { has_holder = yes } }
		e_earth_kingdom = { holder_scope = { save_event_target_as = atx_earth_king } }
	}
	else_if = {
		limit = { k_earth_kingdom = { has_holder = yes } }
		k_earth_kingdom = { holder_scope = { save_event_target_as = atx_earth_king } }
	}
}

# Earth imperialism: the main mod's Earth Kingdom unity law shifts what the states owe (run on the ruler of Ba Sing Se).
# Changing that law needs 100 Royal Authority in the main mod, so it shifts the tiers instead of capping them.
#   Disunion: nobody owes anything.  Territories (default): no obedience.
#   States: as if authority were 15 higher, obedience possible.  Monarchism: 30 higher, but resentment grows every year.
atx_compute_effective_authority_effect = {
	set_variable = { which = global_atx_eff_ra which = global_royalAuthority }
	clr_global_flag = atx_law_allows_obedience
	clr_global_flag = atx_law_states_plus
	clr_global_flag = atx_law_monarchism
	if = {
		limit = { has_law = ek_unity_0 }
		set_variable = { which = global_atx_eff_ra value = -1000 }
	}
	else_if = {
		limit = { has_law = ek_unity_3 }
		change_variable = { which = global_atx_eff_ra value = 30 }
		set_global_flag = atx_law_allows_obedience
		set_global_flag = atx_law_states_plus
		set_global_flag = atx_law_monarchism
	}
	else_if = {
		limit = { has_law = ek_unity_2 }
		change_variable = { which = global_atx_eff_ra value = 15 }
		set_global_flag = atx_law_allows_obedience
		set_global_flag = atx_law_states_plus
	}
	else_if = {
		limit = { NOT = { has_law = ek_unity_1 } } # a crown without unity laws (another government)
		set_global_flag = atx_law_allows_obedience
		set_global_flag = atx_law_states_plus
	}
}

# Budget programmes: ongoing yearly costs while switched on. Unpaid -> the programme lapses (run on the ruler of Ba Sing Se)

}

atx_apply_border_garrisons_effect = {
	any_province = {
		limit = { has_province_modifier = atx_strengthened_border }
		remove_province_modifier = atx_strengthened_border
	}
	any_realm_province = {
		limit = {
			any_neighbor_province = {
				has_owner = yes
				owner = { NOT = { same_realm = ROOT } }
			}
		}
		add_province_modifier = { name = atx_strengthened_border days = 380 }
	}
}

# Dai Li influence (global_atx_daili, 0-100): grows while the crown funds them or is weak, fades otherwise
atx_daili_influence_effect = {
	if = {
		limit = { has_global_flag = atx_prog_daili }
		change_variable = { which = global_atx_daili value = 4 }
	}
	if = {
		limit = { NOT = { check_variable = { which = global_royalAuthority value = 20 } } }
		change_variable = { which = global_atx_daili value = 2 }
	}
	if = {
		limit = {
			NOT = { has_global_flag = atx_prog_daili }
			check_variable = { which = global_royalAuthority value = 20 }
		}
		change_variable = { which = global_atx_daili value = -2 }
	}
	if = {
		limit = { NOT = { check_variable = { which = global_atx_daili value = 0 } } }
		set_variable = { which = global_atx_daili value = 0 }
	}
	if = {
		limit = { check_variable = { which = global_atx_daili value = 100 } }
		set_variable = { which = global_atx_daili value = 100 }
	}
	# Overreach, like Long Feng: the Dai Li hold the court in their hands
	if = {
		limit = {
			check_variable = { which = global_atx_daili value = 75 }
			NOT = { has_character_modifier = atx_daili_ascendant }
		}
		random = {
			chance = 20
			character_event = { id = atlaexp.17 }
		}
	}
}

# Crown works: started by decision (down payment), then one stage a year paid from the treasury.
# A stage that cannot be paid stalls; three stalls in a row abandon the work.
atx_progress_messengers_effect = {
	if = {
		limit = { check_variable = { which = global_atx_w_messengers_stage value = 1 } }
		if = {
			limit = { check_variable = { which = global_atx_treasury value = 200 } }
			change_variable = { which = global_atx_treasury value = -200 }
			change_variable = { which = global_atx_w_messengers_stage value = 1 }
			set_variable = { which = global_atx_w_messengers_stall value = 0 }
			if = {
				limit = { check_variable = { which = global_atx_w_messengers_stage value = 5 } }
				set_variable = { which = global_atx_w_messengers_stage value = 0 }
				atx_complete_messengers_effect = yes
			}
			else = {
				set_character_flag = atx_report_messengers
				character_event = { id = atlaexp.10 }
			}
		}
		else = {
			change_variable = { which = global_atx_w_messengers_stall value = 1 }
			if = {
				limit = { check_variable = { which = global_atx_w_messengers_stall value = 3 } }
				set_variable = { which = global_atx_w_messengers_stage value = 0 }
				set_variable = { which = global_atx_w_messengers_stall value = 0 }
				set_character_flag = atx_abandon_messengers
				character_event = { id = atlaexp.11 }
			}
			else = {
				set_character_flag = atx_stall_messengers
				character_event = { id = atlaexp.11 }
			}
		}
	}
}
